# Creative Editor Route And shadcn/ui Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Split `/editor/:workflowId` out of `/creative`, then migrate the `/creative` main page toward Tailwind CSS and official `shadcn/ui` components.

**Architecture:** `/creative` remains the one-click creation and task status page. `/editor/:workflowId` is a normal route that loads workflow data by route param and renders the existing `CreativeVideoEditor`. Official `shadcn/ui` components are introduced through `shadcn@latest` CLI before `/creative` UI extraction so new shared controls come from the registry instead of local imitation components.

**Tech Stack:** React 19, Vite 8, React Router, Tailwind CSS, `shadcn@latest`, Radix-backed shadcn/ui components, lucide-react, existing Node static tests.

- [ ] **Step 1: Create `CreativeEditorPage.jsx`**

Create `frontend-react/src/pages/CreativeEditorPage.jsx`:

```jsx
import { useCallback, useEffect, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import { ArrowLeft, Loader2 } from 'lucide-react';
import { api } from '../api/client.js';
import { CreativeVideoEditor } from '../components/creative-video-editor/CreativeVideoEditor.jsx';

function getWorkflowPayload(result) {
  return result?.workflow || result?.creative_workflow || result?.data?.workflow || result?.data || result || null;
}

function isNotFoundError(error) {
  return error?.status === 404 || error?.data?.code === 'NOT_FOUND' || error?.data?.error_code === 'NOT_FOUND';
}

function getEditorWorkflowTitle(workflow, workflowId) {
  const rawText = workflow?.creative_context?.input?.raw_text?.trim();
  if (rawText) return rawText.length > 24 ? `${rawText.slice(0, 24)}...` : rawText;
  return workflow?.title || workflowId || '视频编辑器';
}

function getErrorMessage(error) {
  if (isNotFoundError(error)) return '未找到创作任务。';
  return error?.data?.message || error?.message || '加载编辑任务失败，请稍后重试。';
}

export function CreativeEditorPage() {
  const { workflowId = '' } = useParams();
  const navigate = useNavigate();
  const [workflow, setWorkflow] = useState(null);
  const [status, setStatus] = useState(workflowId ? 'loading' : 'missing');
  const [message, setMessage] = useState(workflowId ? '正在加载编辑任务...' : '缺少创作任务 ID。');

  const loadWorkflow = useCallback(async () => {
    if (!workflowId) {
      setWorkflow(null);
      setStatus('missing');
      setMessage('缺少创作任务 ID。');
      return;
    }

    setStatus('loading');
    setMessage('正在加载编辑任务...');
    try {
      const result = await api.getCreativeWorkflow(workflowId);
      const nextWorkflow = getWorkflowPayload(result);
      setWorkflow(nextWorkflow);
      setStatus('ready');
      setMessage('编辑任务已加载。');
    } catch (error) {
      setWorkflow(null);
      setStatus(isNotFoundError(error) ? 'not_found' : 'error');
      setMessage(getErrorMessage(error));
    }
  }, [workflowId]);

  useEffect(() => {
    loadWorkflow();
  }, [loadWorkflow]);

  function returnToCreative() {
    if (workflowId) {
      navigate(`/creative/${encodeURIComponent(workflowId)}`);
      return;
    }
    navigate('/creative');
  }

  const title = getEditorWorkflowTitle(workflow, workflowId);
  const loading = status === 'loading';
  const ready = status === 'ready' && workflow;

  return (
    <main className="creativeEditorPage">
      <header className="creativeEditorHeader">
        <button className="btn secondary" type="button" onClick={returnToCreative}>
          <ArrowLeft size={16} />
          <span>返回创作任务</span>
        </button>
        <div>
          <span>视频编辑器</span>
          <h1>{title}</h1>
        </div>
      </header>

      {!ready ? (
        <section className={`status ${status === 'error' || status === 'not_found' ? 'error' : 'info'}`} aria-live="polite">
          {loading ? <Loader2 size={16} className="spinIcon" /> : null}
          <span>{message}</span>
        </section>
      ) : (
        <CreativeVideoEditor workflowId={workflowId} api={api} onRendered={loadWorkflow} />
      )}
    </main>
  );
}
```

export const STAGE_LABELS = {
  source: '准备来源资料',
  research: '联网研究',
  assets: '素材分析',
  agent_run: '导演改写',

  check: '校验工程',
  render: '渲染视频',
  inspect: '巡检视频',
};

export const DEFAULT_STAGES = Object.entries(STAGE_LABELS).map(([id, label]) => ({
  id,
  label,
  status: 'waiting',
}));

export const STATUS_TEXT = {
  waiting: '等待中',
  pending: '排队中',
  queued: '排队中',
  running: '进行中',
  done: '已完成',
  skipped: '已跳过',
  failed: '失败',
};

export function getStatusClass(status) {
  if (status === 'done' || status === 'skipped') return 'done';
  if (status === 'failed') return 'failed';
  if (status === 'queued' || status === 'pending' || status === 'running') return 'pending';
  return '';
}

export function getStatusMessageClass(status) {
  if (status === 'creating' || status === 'polling') return 'loading';
  if (status === 'done') return 'success';
  if (status === 'failed') return 'error';
  return 'info';
}

export function getWorkflowStatusText(workflow, fallbackStatus = 'idle') {
  const nextStatus = workflow?.status || fallbackStatus;
  if (!workflow && fallbackStatus === 'idle') return '等待输入';
  return STATUS_TEXT[nextStatus] || nextStatus || '等待中';
}

export function normalizeWorkflowStages(workflow) {
  const source = Array.isArray(workflow?.stages) && workflow.stages.length ? workflow.stages : DEFAULT_STAGES;
  return source.map(stage => ({
    ...stage,
    label: stage.label || STAGE_LABELS[stage.id] || stage.id || '未命名阶段',
    status: stage.status || 'waiting',
  }));
}

export function getStepState(stage, index, stages) {
  if (stage.status === 'done') return 'done';
  if (stage.status === 'skipped') return 'done';
  if (stage.status === 'failed') return 'failed';
  if (stage.status === 'running' || stage.status === 'queued' || stage.status === 'pending') return 'active';
  const hasActiveBefore = stages.slice(0, index).some(item => (
    item.status === 'running'
    || item.status === 'queued'
    || item.status === 'pending'
    || item.status === 'failed'
  ));
  return hasActiveBefore ? 'waiting' : '';
}
